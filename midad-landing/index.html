مداد استوديو تصميم وتطوير للشركات التي تريد موقعًا يعمل بقدر ما تعمل. نبدأ من أهداف عملك، ونسلّم موقعًا سريعًا وقابلًا للقياس، ثم نحسّنه بالبيانات كل شهر.

استراتيجية قبل الألوان

نحدد ما يجب أن يحققه الموقع، ثم نصمم كل صفحة لخدمة هذا الهدف.

نظام تصميم، لا صفحات منفصلة

مكوّنات موحّدة تجعل كل صفحة جديدة أسرع وأكثر اتساقًا.

أداء نضمنه بالأرقام

نطلق بدرجات أداء عالية على الجوال، ونتابع التحويل بعد الإطلاق.

import { getFeatured } from "@/lib/catalog";

export const revalidate = 60;

export default async function Home() {
  const products = await getFeatured({ limit: 8 });
  return (
    <main dir="rtl">
      <Hero title="عطور تُروى حكايتها" cta="تسوّق المجموعة" />
      <section className="grid">
        {products.map((p) => <ProductCard key={p.id} product={p} />)}
      </section>
    </main>
  );
}
