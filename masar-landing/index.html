// Salla order → AI triage → WhatsApp
{
  "name": "salla-new-order",
  "nodes": [
    { "type": "n8n-nodes-base.webhook", "path": "salla/order.created" },
    { "type": "@n8n/n8n-nodes-langchain.agent",
      "prompt": "Classify order: standard | vip | risk" },
    { "type": "n8n-nodes-base.whatsApp", "template": "order_confirmed_ar" },
    { "type": "n8n-nodes-base.googleSheets", "operation": "appendOrUpdate" }
  ],
  "settings": { "retryOnFail": true, "maxTries": 3 }
}
# Trigger the workflow from any system
curl -X POST https://hooks.masar.sa/w/salla-new-order \
  -H "Authorization: Bearer $MASAR_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "order_id": 48213,
    "customer": { "name": "عبدالله", "phone": "+9665XXXXXXXX" },
    "total": 386.50,
    "currency": "SAR"
  }'

# → 202 Accepted  { "run_id": "run_8f2c1a", "status": "queued" }
from masar import Agent, tools

support = Agent(
    name="customer-support-ar",
    knowledge=["policies/returns.md", "catalog.csv"],
    tools=[tools.salla.orders, tools.whatsapp.send],
    require_approval=["refund", "discount_over_15"],
)

reply = support.run("وين طلبي؟ رقمه 48213")
print(reply.text, reply.actions)
